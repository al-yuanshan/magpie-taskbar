cmake_minimum_required(VERSION 3.20)
project(magpie-taskbar VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(route_model INTERFACE)
target_include_directories(route_model INTERFACE src third_party)
include(CTest)
if(BUILD_TESTING)
    add_executable(route_tests tests/route_tests.cpp)
    target_link_libraries(route_tests PRIVATE route_model)
    if(MSVC)
        target_compile_options(route_tests PRIVATE /utf-8)
    endif()
    add_test(NAME route_tests COMMAND route_tests)
endif()
if(WIN32)
    if(NOT MSVC)
        message(FATAL_ERROR "Use MSVC to match TrafficMonitor's C++ plugin ABI")
    endif()
    add_library(magpie-taskbar SHARED src/magpie_plugin.cpp src/http.cpp src/options_dialog.cpp src/magpie_taskbar.def)
    target_link_libraries(magpie-taskbar PRIVATE route_model winhttp shell32 comctl32 user32)
    target_compile_definitions(magpie-taskbar PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
    target_compile_options(magpie-taskbar PRIVATE /W4 /utf-8 /EHsc)
    set_property(TARGET magpie-taskbar PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    install(TARGETS magpie-taskbar RUNTIME DESTINATION .)
endif()
